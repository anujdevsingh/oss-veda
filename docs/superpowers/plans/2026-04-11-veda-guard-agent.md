> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a guard agent that runs 14 automated checks (pre-flight + security + post-mortem) before and after the oss-veda pipeline, blocking on hard failures and warning on soft issues.

**Architecture:** A PEP 723 Python script (`guard.py`, stdlib only) handles all deterministic checks and outputs JSON. A thin Claude subagent (`veda-guard.md`) orchestrates the script and translates results to plain English. The guard is wired into SKILL.md and veda-deep.md as Step 0 (pre-flight) and Step 4 (post-mortem).

**Tech Stack:** Python 3.11+ stdlib (hashlib, subprocess, socket, json, pathlib, tempfile, argparse, sys, os, re, time), PEP 723 inline metadata, Claude Code agent markdown format.

```python
#!/usr/bin/env python3
"""Dev-only: regenerate checksums.json from current script files."""

import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

SCRIPTS_DIR = Path(__file__).parent
PLUGIN_ROOT = SCRIPTS_DIR.parent
CHECKSUMS_FILE = PLUGIN_ROOT / "checksums.json"

# Read version from plugin.json
plugin_json = json.loads((PLUGIN_ROOT / ".claude-plugin" / "plugin.json").read_text())
version = plugin_json["version"]

# Hash all .py files in scripts/ (excluding this file and __pycache__)
files = {}
for py_file in sorted(SCRIPTS_DIR.glob("*.py")):
    if py_file.name.startswith("_"):
        continue
    sha256 = hashlib.sha256(py_file.read_bytes()).hexdigest()
    files[f"scripts/{py_file.name}"] = sha256

checksums = {
    "version": version,
    "generated_at": datetime.now(timezone.utc).isoformat(),
    "algorithm": "sha256",
    "files": files,
}

CHECKSUMS_FILE.write_text(json.dumps(checksums, indent=2) + "\n")
print(f"Wrote {len(files)} checksums to {CHECKSUMS_FILE}")
for path, sha in files.items():
    print(f"  {sha[:12]}...  {path}")
```

```python
def make_check(check_id: str, name: str, severity: str) -> dict:
    """Create a check result dict."""
    return {
        "id": check_id,
        "name": name,
        "status": "pass",
        "severity": severity,
        "details": "",
        "fix_hint": None,
    }


def check_uv_installed() -> dict:
    """Check 1: uv is installed and on PATH."""
    c = make_check("uv_installed", "uv installed", "hard")
    try:
        result = subprocess.run(
            ["uv", "--version"],
            capture_output=True, text=True, timeout=10,
        )
        if result.returncode == 0:
            version = result.stdout.strip()
            c["details"] = f"{version} found"
        else:
            c["status"] = "fail"
            c["details"] = "uv found but returned an error"
            c["fix_hint"] = (
                "Reinstall uv: https://docs.astral.sh/uv/getting-started/installation/"
            )
    except FileNotFoundError:
        c["status"] = "fail"
        c["details"] = "uv is not installed or not on PATH"
        c["fix_hint"] = (
            "Install uv:\n"
            "  macOS/Linux:  curl -LsSf https://astral.sh/uv/install.sh | sh\n"
            "  Windows:      powershell -c \"irm https://astral.sh/uv/install.ps1 | iex\"\n"
            "  Docs:         https://docs.astral.sh/uv/getting-started/installation/\n"
            "Then restart Claude Code and re-run /veda."
        )
    except subprocess.TimeoutExpired:
        c["status"] = "fail"
        c["details"] = "uv --version timed out after 10s"
        c["fix_hint"] = "uv may be hung. Try running 'uv --version' in your terminal."
    return c
```

```python
def check_cache_size() -> dict:
    """Check 8: Cache dir is not unreasonably large (< 100 MB)."""
    c = make_check("cache_size", "Cache dir size sane", "soft")
    if not CACHE_DIR.exists():
        c["details"] = "Cache dir does not exist yet (first run)"
        return c
    total_bytes = sum(
        f.stat().st_size for f in CACHE_DIR.rglob("*") if f.is_file()
    )
    total_mb = total_bytes / (1024 * 1024)
    c["details"] = f"Cache size: {total_mb:.1f} MB"
    if total_mb > 100:
        c["status"] = "warn"
        c["fix_hint"] = (
            f"Cache dir is {total_mb:.0f} MB (threshold: 100 MB).\n"
            "Auto-cleaning old cache files..."
        )
        # Auto-fix: remove oldest files until under 80 MB
        files_by_age = sorted(
            [f for f in CACHE_DIR.rglob("*") if f.is_file()],
            key=lambda f: f.stat().st_mtime,
        )
        while total_mb > 80 and files_by_age:
            oldest = files_by_age.pop(0)
            size = oldest.stat().st_size
            try:
                oldest.unlink()
                total_mb -= size / (1024 * 1024)
            except OSError:
                pass
        c["details"] = f"Cache cleaned to {total_mb:.1f} MB"
    return c


def check_unexpected_files() -> dict:
    """Check 9: No unexpected files in cache dir."""
    c = make_check("unexpected_files", "No unexpected cache files", "soft")
    if not CACHE_DIR.exists():
        c["details"] = "Cache dir does not exist yet"
        return c
    expected_patterns = [
        r"^scouts_[a-zA-Z0-9_-]+_\d+d\.json$",
        r"^oss-veda-raw\.json$",
        r"^oss-veda-ranked\.json$",
        r"^oss-veda-report\.md$",
        r"^run_history\.tsv$",
        r"^\.guard_write_test$",
    ]
    unexpected = []
    for f in CACHE_DIR.iterdir():
        if f.is_file():
            if not any(re.match(p, f.name) for p in expected_patterns):
                unexpected.append(f.name)
    if unexpected:
        c["status"] = "warn"
        c["details"] = f"Unexpected files in cache: {', '.join(unexpected[:5])}"
        c["fix_hint"] = (
            "The cache directory contains files not created by oss-veda.\n"
            f"Location: {CACHE_DIR}\n"
            "This is probably harmless but worth investigating."
        )
    else:
        c["details"] = "Only expected files in cache dir"
    return c


def check_script_checksums() -> dict:
    """Check 10: Script SHA256 checksums match checksums.json."""
    c = make_check("script_checksums", "Script checksums match", "hard")
    checksums_path = PLUGIN_ROOT / "checksums.json"
    if not checksums_path.exists():
        c["status"] = "warn"
        c["severity"] = "soft"  # Downgrade: missing checksums file shouldn't block
        c["details"] = "checksums.json not found — cannot verify script integrity"
        c["fix_hint"] = "Reinstall the plugin to restore checksums.json: /plugin update oss-veda"
        return c
    try:
        with open(checksums_path, encoding="utf-8") as f:
            checksums_data = json.load(f)
    except (json.JSONDecodeError, OSError) as e:
        c["status"] = "warn"
        c["severity"] = "soft"
        c["details"] = f"checksums.json is corrupted: {e}"
        c["fix_hint"] = "Reinstall the plugin: /plugin update oss-veda"
        return c

    expected_files = checksums_data.get("files", {})
    mismatches = []
    for rel_path, expected_hash in expected_files.items():
        full_path = PLUGIN_ROOT / rel_path
        if not full_path.exists():
            mismatches.append(f"{rel_path}: MISSING")
            continue
        actual_hash = hashlib.sha256(full_path.read_bytes()).hexdigest()
        if actual_hash != expected_hash:
            mismatches.append(
                f"{rel_path}: expected {expected_hash[:12]}... got {actual_hash[:12]}..."
            )

    if mismatches:
        c["status"] = "fail"
        c["details"] = f"Checksum mismatch: {'; '.join(mismatches)}"
        c["fix_hint"] = (
            "One or more plugin scripts have been modified.\n"
            "This could mean:\n"
            "  (a) You modified a file yourself for debugging.\n"
            "  (b) Something else modified the file, which is concerning.\n"
            "Fix: /plugin update oss-veda  (restores original files)"
        )
    else:
        c["details"] = f"All {len(expected_files)} scripts verified"
    return c


def check_history_sane() -> dict:
    """Check 11: run_history.tsv is not corrupted."""
    c = make_check("history_sane", "run_history.tsv sane", "soft")
    history_path = CACHE_DIR / "run_history.tsv"
    if not history_path.exists():
        c["details"] = "No run history yet (first run)"
        return c
    try:
        content = history_path.read_text(encoding="utf-8")
        lines = content.splitlines()
        if not lines:
            c["details"] = "run_history.tsv is empty"
            return c
        # Check header
        expected_header = "timestamp\ttopic\tdays\tscouts_ok\telapsed_s\trepos\tissues\thn\treddit\thf\tpwc"
        if lines[0].strip() != expected_header:
            c["status"] = "warn"
            c["details"] = "run_history.tsv has unexpected header"
            c["fix_hint"] = "Auto-fix: truncating corrupted history file."
            # Auto-fix: rewrite with correct header
            history_path.write_text(expected_header + "\n", encoding="utf-8")
            return c
        # Check for oversized rows (>10KB = probably binary junk)
        oversized = [i for i, line in enumerate(lines) if len(line) > 10240]
        if oversized:
            c["status"] = "warn"
            c["details"] = f"run_history.tsv has {len(oversized)} oversized row(s)"
            c["fix_hint"] = "Auto-fix: removing corrupted rows."
            clean_lines = [l for l in lines if len(l) <= 10240]
            history_path.write_text("\n".join(clean_lines) + "\n", encoding="utf-8")
        else:
            c["details"] = f"run_history.tsv OK ({len(lines) - 1} entries)"
    except OSError as e:
        c["status"] = "warn"
        c["details"] = f"Cannot read run_history.tsv: {e}"
    return c
```

```python
# --- Post-mortem checks ---

FAILURE_PATTERNS = [
    (["TimeoutError", "asyncio.TimeoutError", "timed out"], "timeout",
     "Scout exceeded the 3-minute timeout. The API was likely slow or unresponsive.",
     "No action needed — try again later."),
    (["403", "429", "rate limit", "Rate limit"], "rate_limit",
     "Hit API rate limit. Backoff did not recover in time.",
     "Wait 5 minutes before re-running, or set GITHUB_TOKEN for higher limits."),
    (["ConnectionError", "getaddrinfo failed", "Network unreachable", "ConnectError"], "network",
     "Cannot reach the API host. Network may be down or blocked.",
     "Check your internet connection."),
    (["JSONDecodeError", "Expecting value", "Invalid JSON"], "bad_response",
     "API returned non-JSON (likely an HTML error page or maintenance notice).",
     "Usually transient — try again in a few minutes."),
    (["KeyError", "AttributeError", "TypeError"], "api_change",
     "API response shape may have changed. This is likely a plugin bug.",
     "Report this at https://github.com/anujdevsingh/oss-veda/issues"),
]


def classify_failure(error_str: str) -> tuple[str, str, str]:
    """Classify a scout failure by matching error patterns.
    Returns (category, diagnosis, user_action)."""
    for patterns, category, diagnosis, action in FAILURE_PATTERNS:
        if any(p.lower() in error_str.lower() for p in patterns):
            return category, diagnosis, action
    return "unknown", f"Unrecognized error: {error_str[:200]}", (
        "Report this at https://github.com/anujdevsingh/oss-veda/issues with the full error."
    )


def check_scouts_succeeded(raw_data: dict) -> dict:
    """Check 12: How many scouts succeeded."""
    c = make_check("scouts_succeeded", "Scouts succeeded", "soft")
    meta = raw_data.get("_metadata", {})
    succeeded = meta.get("scouts_succeeded", 0)
    total = meta.get("scouts_total", 5)
    c["details"] = f"{succeeded}/{total} scouts succeeded"
    if succeeded <= 2:
        c["status"] = "fail"
        c["severity"] = "hard"
        c["fix_hint"] = f"Only {succeeded}/{total} scouts returned data. Report may be unreliable."
    elif succeeded < total:
        c["status"] = "warn"
        c["details"] = f"{succeeded}/{total} scouts succeeded ({total - succeeded} failed)"
    return c


def check_report_exists() -> dict:
    """Check 13: Report file was generated and is non-trivial."""
    c = make_check("report_exists", "Report file generated", "hard")
    report_path = Path(tempfile.gettempdir()) / "oss-veda-report.md"
    if not report_path.exists():
        c["status"] = "fail"
        c["details"] = "oss-veda-report.md not found"
        c["fix_hint"] = "The report generation step may have failed. Check for errors above."
        return c
    size = report_path.stat().st_size
    if size < 100:
        c["status"] = "fail"
        c["details"] = f"Report file is only {size} bytes (expected >100)"
        c["fix_hint"] = "Report is suspiciously small. Generation may have failed."
    else:
        c["details"] = f"Report OK ({size:,} bytes)"
    return c


def check_no_leaked_tokens() -> dict:
    """Check 14: No API tokens leaked in the report output."""
    c = make_check("no_leaked_tokens", "No tokens in report", "hard")
    report_path = Path(tempfile.gettempdir()) / "oss-veda-report.md"
    if not report_path.exists():
        c["details"] = "No report to check (skipped)"
        return c
    try:
        content = report_path.read_text(encoding="utf-8")
    except OSError:

    leak_patterns = ["ghp_", "github_pat_", "Bearer ", "gho_", "ghu_"]
    found = [p for p in leak_patterns if p in content]
    if found:
        c["status"] = "fail"
        c["details"] = f"Possible token leak in report: patterns {found} found"
        c["fix_hint"] = (
            "The report may contain an API token. Do NOT share this report.\n"
            "This should never happen — please report it at\n"
            "https://github.com/anujdevsingh/oss-veda/issues"
        )
    else:
        c["details"] = "No token patterns found in report"
    return c
```

```python
def diagnose_failures(raw_data: dict) -> list[dict]:
    """Analyze raw scout data and classify any failures."""
    failures = []
    scout_names = ["github", "hackernews", "reddit", "huggingface", "papers_with_code"]
    for name in scout_names:
        data = raw_data.get(name)
        if data is None:
            # Scout key missing entirely — likely crashed before producing output
            failures.append({
                "scout": name,
                "category": "unknown",
                "raw_error": "No data returned (scout may have crashed)",
                "diagnosis": f"The {name} scout produced no output.",
                "user_action": "Try re-running. If persistent, report the issue.",
            })
        elif isinstance(data, list) and len(data) == 0:
            # Empty results — not necessarily a failure (could just be quiet day)
            pass  # Don't flag empty results as failures
    return failures
```

```python
def run_preflight() -> dict:
    """Run all 11 pre-flight + security checks."""
    start = time.time()
    checks = [
        check_uv_installed(),
        check_python_version(),
        check_temp_dir_writable(),
        check_network_reachable(),
        check_github_token(),
        check_scripts_exist(),
        check_config_valid(),
        check_cache_size(),
        check_unexpected_files(),
        check_script_checksums(),
        check_history_sane(),
    ]
    elapsed = round(time.time() - start, 2)

    failed = [c for c in checks if c["status"] == "fail"]
    warned = [c for c in checks if c["status"] == "warn"]
    passed = [c for c in checks if c["status"] == "pass"]

    if failed:
        status = "hard_fail"
    elif warned:
        status = "warnings"
    else:
        status = "ok"

    return {
        "mode": "preflight",
        "status": status,
        "duration_seconds": elapsed,
        "checks": checks,
        "summary": {
            "passed": len(passed),
            "warned": len(warned),
            "failed": len(failed),
            "total": len(checks),
        },
    }


def run_postmortem(input_path: str) -> dict:
    """Run post-mortem checks on pipeline output."""
    raw_path = Path(input_path)
    if not raw_path.exists():
        return {
            "mode": "postmortem",
            "error": f"Input file not found: {input_path}",
            "scouts_succeeded": 0,
            "scouts_total": 5,
            "failures": [],
            "leaked_tokens_in_report": False,
        }

    with open(raw_path, encoding="utf-8") as f:
        raw_data = json.load(f)

    meta = raw_data.get("_metadata", {})
    checks = [
        check_scouts_succeeded(raw_data),
        check_report_exists(),
        check_no_leaked_tokens(),
    ]
    failures = diagnose_failures(raw_data)

    # Also classify any error info embedded in the raw data
    # (run_scouts.py stores exceptions as empty lists, but stderr has the info)
    # For now, we rely on _metadata.scouts_succeeded count

    leaked = any(c["id"] == "no_leaked_tokens" and c["status"] == "fail" for c in checks)

    return {
        "mode": "postmortem",
        "scouts_succeeded": meta.get("scouts_succeeded", 0),
        "scouts_total": meta.get("scouts_total", 5),
        "checks": checks,
        "failures": failures,
        "leaked_tokens_in_report": leaked,
    }


def main():
    parser = argparse.ArgumentParser(description="oss-veda guard")
    parser.add_argument(
        "--mode", choices=["preflight", "postmortem"], default="preflight",
        help="Check mode: preflight (before pipeline) or postmortem (after pipeline)",
    )
    parser.add_argument(
        "--input", default=str(Path(tempfile.gettempdir()) / "oss-veda-raw.json"),
        help="Input file for postmortem mode (default: temp dir oss-veda-raw.json)",
    )
    args = parser.parse_args()

    if args.mode == "preflight":
        result = run_preflight()
    else:
        result = run_postmortem(args.input)

    json.dump(result, sys.stdout, indent=2)
    print()  # trailing newline

    # Exit code: 0 = ok/warnings, 1 = hard_fail
    if result.get("status") == "hard_fail":
        sys.exit(1)
    if result.get("error"):
        sys.exit(1)


if __name__ == "__main__":
    main()
```

Every `/veda` and `/veda-deep` run starts with an automatic guard check (~3 seconds). The guard verifies your environment is healthy and safe before running the pipeline.
